  FUNCTION MpMsg(Tag:string, Message:string, Type:number)
    IF Multiplayer.is-host()
      IF System.compare-eventvar(variable=Type, comparison=0.0, value=0)
        -> Multiplayer.broadcast-message(from-id=Multiplayer.MyID, tag=Tag, message=Message, mode=reliable-ordered)
      IF System.compare-eventvar(variable=Type, comparison=0.0, value=1)
        -> Multiplayer.broadcast-message(from-id=Multiplayer.MyID, tag=Tag, message=Message, mode=reliable-unordered)
      IF System.compare-eventvar(variable=Type, comparison=0.0, value=2)
        -> Multiplayer.broadcast-message(from-id=Multiplayer.MyID, tag=Tag, message=Message, mode=unreliable)
    IF NOT Multiplayer.is-host()
      IF System.compare-eventvar(variable=Type, comparison=0.0, value=0)
        -> Multiplayer.send-message(peer-id=Multiplayer.HostID, tag=Tag, message=Message, mode=reliable-ordered)
      IF System.compare-eventvar(variable=Type, comparison=0.0, value=1)
        -> Multiplayer.send-message(peer-id=Multiplayer.HostID, tag=Tag, message=Message, mode=reliable-unordered)
      IF System.compare-eventvar(variable=Type, comparison=0.0, value=2)
        -> Multiplayer.send-message(peer-id=Multiplayer.HostID, tag=Tag, message=Message, mode=unreliable)
  // Auto broadcast
  IF Multiplayer.on-any-peer-message() AND Multiplayer.is-host()
