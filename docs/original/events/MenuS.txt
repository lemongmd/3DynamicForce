    -> MenuNick.set-max-length(max-length=12)
    -> MenuNick.set-text(text=GamePush.PlayerName)
    IF NOT Audio.is-tag-playing(tag="menumusic")
      -> Audio.stop(tag="music")
      -> Audio.play(audio-file={ "path": "Menu_loop" }, loop=looping, volume=0, stereo-pan=0, tag-optional="music menumusic m")
  ALWAYS
    -> call Set text(GamePush.PlayerScore & "🍩", "Muffins", false)
  IF Multiplayer.supports-multiplayer() AND NOT Multiplayer.is-connected()
    -> Multiplayer.connect(server="wss://multiplayer.scirra.com")
  IF Multiplayer.is-in-room()
    -> MenuConnectionStatus.set-animation-frame(frame-number=5)
    IF Text.compare-instance-variable(instance-variable=Id, comparison=0.0, value="SS")
      -> Text.set-text(text="подключение к комнате...")
  IF System.else() AND Multiplayer.is-logged-in()
    -> MenuConnectionStatus.set-animation-frame(frame-number=4)
    IF Text.compare-instance-variable(instance-variable=Id, comparison=0.0, value="SS")
      -> Text.set-text(text="выполнен вход")
  IF System.else() AND Multiplayer.is-connected()
    -> MenuConnectionStatus.set-animation-frame(frame-number=3)
    IF Text.compare-instance-variable(instance-variable=Id, comparison=0.0, value="SS")
      -> Text.set-text(text="подключено")
  IF System.else() AND Multiplayer.supports-multiplayer()
    -> MenuConnectionStatus.set-animation-frame(frame-number=2)
    IF Text.compare-instance-variable(instance-variable=Id, comparison=0.0, value="SS")
      -> Text.set-text(text="подключение...")
  IF System.else()
    -> MenuConnectionStatus.set-animation-frame(frame-number=1)
    IF Text.compare-instance-variable(instance-variable=Id, comparison=0.0, value="SS")
      -> Text.set-text(text="нет сети!!!")
  IF Touch.on-touched-object(object=MenuStartButton, type=start)
    -> Audio.play(audio-file={ "path": "Click" }, loop=not-looping, volume=0, stereo-pan=0, tag-optional="")
    // Play button
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=0)
      IF Multiplayer.is-connected() AND NOT Multiplayer.is-logged-in()
        -> call Change page("MenuStart", "MenuMap", 208, 104)
      IF System.else()
        -> System.set-eventvar-value(variable=nco, value=100)
        IF System.compare-two-values(first-value=floor(random(100)), comparison=0.0, second-value=2)
          -> Audio.play(audio-file={ "path": "burp" }, loop=not-looping, volume=0, stereo-pan=0, tag-optional="sfx")
        IF System.else()
          -> Audio.play(audio-file={ "path": "Cancel" }, loop=not-looping, volume=0, stereo-pan=0, tag-optional="sfx")
    // To skin page
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=1)
      -> call Change page("MenuStart", "MenuSkin", 240, 64)
    // To credits
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=2)
      -> call Fade to layout("Credits", false)
    // From skin back to main
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=3)
      -> call Change page("MenuSkin", "MenuStart", 168, 72)
      -> GamePush.player-sync(override=false, storage=preferred)
    // From map back to main (log out)
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=4)
      -> call Change page("MenuMap", "MenuStart", 168, 72)
      -> Multiplayer.disconnect-signalling()
    // Achievements
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=5)
      -> GamePush.achievements-open()
    // Leaderboard (unused)
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=6)
      -> GamePush.leaderboard-scoped-open(id-or-tag="muff", variant="score", order=default, limit=0, with-me=me-firts, include-fields="score", display-fields="score", showNearest=0)
    // Site
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=7)
      -> Browser.open-url-in-new-window(url="https://macestudios.ru/", tag="NewWindow")
    // Auth
    IF MenuStartButton.compare-animation-frame(comparison=0.0, number=8)
  var nco: number = 0
  IF System.compare-eventvar(variable=nco, comparison=4.0, value=0) AND Text.compare-instance-variable(instance-variable=Id, comparison=0.0, value="noconnection")
    -> System.subtract-from-eventvar(variable=nco, value=dt * 50)
    -> Text.set-opacity(opacity=nco)
  FUNCTION Change page(From:string, To:string, PNW:number, PNH:number)
    -> System.wait(seconds=0, use-timescale=true)
    -> System.set-layer-visible(layer=From, visibility=invisible)
    -> System.set-layer-interactive(layer=From, interactive=false)
    -> System.set-layer-visible(layer=To, visibility=visible)
    -> System.set-layer-interactive(layer=To, interactive=true)
    -> MenuPanelBg.Tween.tween-two-properties(tags="size", property=size, end-x=PNW, end-y=PNH, time=1 / 6, ease=easeoutback, destroy-on-complete=no, loop=no, ping-pong=no, repeat-count=1)
  IF System.layer-is-visible(layer="MenuMap") AND NOT Multiplayer.is-logged-in()
    -> Multiplayer.log-in(alias=(GamePush.PlayerName = "" ? right(GamePush.PlayerID & "", 5) : GamePush.PlayerName ) &";;"& GamePush.PlayerGet("skin"))
  // Join room
  IF Touch.on-touched-object(object=MenuSelectMap, type=start)
    IF Multiplayer.is-logged-in()
      -> Multiplayer.auto-join-room(game=Gamename, instance=tokenat("abcdefghijkl", MenuSelectMap.AnimationFrame, "") & ";" & projectversion, first-room="room", max-peers=4, locking=dont-lock)
  IF Multiplayer.is-in-room()
    -> call Fade to layout("Map" & tokenat(Multiplayer.CurrentInstance, 0, ";"), false)
    IF System.trigger-once-while-true()
      -> Audio.play(audio-file={ "path": "Chosen" }, loop=not-looping, volume=0, stereo-pan=0, tag-optional="sfx")
== GROUP MenuSkin ==
  var SkinAll: string = None
  var SkinR: number = 3
  var SkinG: number = 3
  var SkinB: number = 3
  var SkinBase: number = 0
  var SkinGlass: number = 0
  var SkinPrint: number = 0
  var SkinWheels: number = 0
  IF System.on-start-of-layout()
    -> System.create-object(object-to-create=Player, layer="MenuSkin", x=192, y=136, create-hierarchy=true, template-name="Player")
    -> Player.set-angle(angle=-90)
    -> Player.Car.set-enabled(state=disabled)
    var tempskin: string = 
    ALWAYS
      -> System.set-eventvar-value(variable=tempskin, value=GamePush.PlayerGet("skin"))
      -> Player.set-default-color(color=rgbEx255(int(tokenat(tempskin, 0, "")) * 85, int(tokenat(tempskin, 1, "")) * 85, int(tokenat(tempskin, 2, "")) * 85))
      -> System.set-eventvar-value(variable=SkinR, value=int(tokenat(tempskin, 0, "")))
      -> System.set-eventvar-value(variable=SkinG, value=int(tokenat(tempskin, 1, "")))
      -> System.set-eventvar-value(variable=SkinB, value=int(tokenat(tempskin, 2, "")))
      -> Player.set-animation-frame(frame-number=int(tokenat(tempskin, 4, "")))
      -> System.set-eventvar-value(variable=SkinBase, value=int(tokenat(tempskin, 4, "")))
      IF PlayerSkinPart.is-animation-playing(animation="Glass")
        -> PlayerSkinPart.set-animation-frame(frame-number=int(tokenat(tempskin, 5, "")))
        -> System.set-eventvar-value(variable=SkinGlass, value=int(tokenat(tempskin, 5, "")))
      IF PlayerSkinPart.is-animation-playing(animation="Print")
        -> PlayerSkinPart.set-animation-frame(frame-number=int(tokenat(tempskin, 6, "")))
        -> System.set-eventvar-value(variable=SkinPrint, value=int(tokenat(tempskin, 6, "")))
      IF PlayerSkinPart.is-animation-playing(animation="Wheels")
        -> PlayerSkinPart.set-animation-frame(frame-number=int(tokenat(tempskin, 7, "")))
        -> System.set-eventvar-value(variable=SkinWheels, value=int(tokenat(tempskin, 7, "")))
      IF MenuSkinChange.is-animation-playing(animation="R")
        -> MenuSkinChange.set-animation-frame(frame-number=SkinR)
      IF MenuSkinChange.is-animation-playing(animation="G")
        -> MenuSkinChange.set-animation-frame(frame-number=SkinG)
      IF MenuSkinChange.is-animation-playing(animation="B")
        -> MenuSkinChange.set-animation-frame(frame-number=SkinB)
  IF GamePush.is-player-logged-in() AND System.trigger-once-while-true()
    -> MenuNick.set-text(text=GamePush.PlayerGet("name"))
    -> System.set-eventvar-value(variable=SkinAll, value=GamePush.PlayerGet("skin"))
  IF MenuNick.on-text-changed()
    -> MenuNick.set-text(text=replace(
replace(
replace(
replace(
lowercase(Self.Text)
, "хентай", "honda")
, "rule34", "top")
, "nick:bulava", "ABCDE")
, "nick:lmgd", "FGHI")
)
    -> GamePush.player-set-name(name=MenuNick.Text)
  // Skin
  IF Touch.on-touched-object(object=MenuSkinChange, type=start)
    IF MenuSkinChange.is-animation-playing(animation="R")
      -> System.set-eventvar-value(variable=SkinR, value=(SkinR + 1) % 4)
      -> MenuSkinChange.set-animation-frame(frame-number=SkinR)
    IF MenuSkinChange.is-animation-playing(animation="G")
      -> System.set-eventvar-value(variable=SkinG, value=(SkinG + 1) % 4)
      -> MenuSkinChange.set-animation-frame(frame-number=SkinG)
    IF MenuSkinChange.is-animation-playing(animation="B")
      -> System.set-eventvar-value(variable=SkinB, value=(SkinB + 1) % 4)
      -> MenuSkinChange.set-animation-frame(frame-number=SkinB)
    IF MenuSkinChange.is-animation-playing(animation="SkinParts")
      IF MenuSkinChange.compare-animation-frame(comparison=0.0, number=0)
        -> Player.set-animation-frame(frame-number=(Self.AnimationFrame + 1) % (Self.AnimationFrameCount - 1))
        -> System.set-eventvar-value(variable=SkinBase, value=Player.AnimationFrame)
      IF MenuSkinChange.compare-animation-frame(comparison=0.0, number=1) AND PlayerSkinPart.is-animation-playing(animation="Glass")
        -> PlayerSkinPart.set-animation-frame(frame-number=(Self.AnimationFrame + 1) % Self.AnimationFrameCount)
        -> System.set-eventvar-value(variable=SkinGlass, value=PlayerSkinPart.AnimationFrame)
      IF MenuSkinChange.compare-animation-frame(comparison=0.0, number=2) AND PlayerSkinPart.is-animation-playing(animation="Print")
        -> PlayerSkinPart.set-animation-frame(frame-number=(Self.AnimationFrame + 1) % Self.AnimationFrameCount)
        -> System.set-eventvar-value(variable=SkinPrint, value=PlayerSkinPart.AnimationFrame)
      IF MenuSkinChange.compare-animation-frame(comparison=0.0, number=3) AND PlayerSkinPart.is-animation-playing(animation="Wheels")
        -> PlayerSkinPart.set-animation-frame(frame-number=(Self.AnimationFrame + 1) % Self.AnimationFrameCount)
        -> System.set-eventvar-value(variable=SkinWheels, value=PlayerSkinPart.AnimationFrame)
    ALWAYS
      -> Player.set-default-color(color=rgbEx255(SkinR * 85, SkinG * 85, SkinB * 85))
      -> System.set-eventvar-value(variable=SkinAll, value="" & SkinR & SkinG & SkinB & 0 & SkinBase & SkinGlass & SkinPrint & SkinWheels)
      -> GamePush.player-set(key="skin", value=SkinAll)
      -> (disabled) GamePush.player-set(key="deanon", value=Browser.UserAgent)
  // Tank
  IF Touch.on-hold-over-object(object=MenuSkinChange) AND MenuSkinChange.is-animation-playing(animation="SkinParts") AND MenuSkinChange.compare-animation-frame(comparison=0.0, number=0)
    -> Player.set-animation-frame(frame-number=Self.AnimationFrameCount)
    -> System.set-eventvar-value(variable=SkinBase, value=Player.AnimationFrame)
